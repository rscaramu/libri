Capitolo 39 – Equazioni differenziali ordinarie

algoritmo EULERO(f, t0, y0, h, n)
    t <- t0
    y <- y0
    per i da 1 a n
        y <- y + h * f(t, y)
        t <- t0 + i * h
    ritorna y

algoritmo PASSO_HEUN(f, t, y, h)
    k1 <- f(t, y)
    k2 <- f(t + h, y + h * k1)
    ritorna y + h * (k1 + k2) / 2

algoritmo PASSO_RK4(f, t, y, h)
    k1 <- f(t, y)
    k2 <- f(t + h/2, y + h/2 * k1)
    k3 <- f(t + h/2, y + h/2 * k2)
    k4 <- f(t + h, y + h * k3)
    ritorna y + h * (k1 + 2*k2 + 2*k3 + k4) / 6

algoritmo PASSO_SEMI_IMPLICITO(x, v, accelerazione, h)
    v <- v + h * accelerazione(x)
    x <- x + h * v                  // usa la velocità nuova
    ritorna (x, v)

algoritmo INTEGRA_ADATTIVO(f, t, y, t_fine, h, tolleranza)
    finché t < t_fine
        h <- min(h, t_fine - t)
        intero <- PASSO_RK4(f, t, y, h)
        mezzo <- PASSO_RK4(f, t, y, h/2)
        doppio <- PASSO_RK4(f, t + h/2, mezzo, h/2)
        errore <- |doppio - intero|
        se errore ≤ tolleranza allora
            t <- t + h                  // passo accettato
            y <- doppio
        // nuovo passo: l'errore di RK4 va come h^5
        h <- 0,9 * h * (tolleranza / errore)^(1/5)
    ritorna y
